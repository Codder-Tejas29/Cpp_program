#include <iostream>

struct Node {
    int data;
    Node* left;
    Node* right;

    Node(int val) {
        data = val;
        left = nullptr;
        right = nullptr;
    }
};

Node* insert(Node* root, int val) {
    
    if (root == nullptr) {
        return new Node(val);
    }
    if (val < root->data) {
        root->left = insert(root->left, val);
    } else if (val > root->data) {
        root->right = insert(root->right, val);
    }
    return root;
}

void inorder(Node* root) {
    if (root == nullptr) return;
    inorder(root->left);
    cout << root->data << " ";
    inorder(root->right);
}

int longestPath(Node* root) {
    if (root == nullptr) {
        return 0;
    }
    int leftHeight = longestPath(root->left);
    int rightHeight = longestPath(root->right);
    return max(leftHeight, rightHeight) + 1;
}

int findMin(Node* root) {
    if (root == nullptr) {
        cout << "Tree is empty!" << endl;
        return -1;
    }
    Node* current = root;
    while (current->left != nullptr) {
        current = current->left; 
    }
    return current->data;
}
void mirrorTree(Node* root) {
    if (root == nullptr) {
        return;
    }
    
    Node* temp = root->left;
    root->left = root->right;
    root->right = temp;
    
    mirrorTree(root->left);
    mirrorTree(root->right);
}

bool searchBST(Node* root, int key) {
    if (root == nullptr) {
        return false;
    }
    if (root->data == key) {
        return true;
    }
    if (key > root->data) {
        return searchBST(root->left, key);
    } else {
        return searchBST(root->right, key);
    }
}

struct Node
{
    string name;
    string phone;
    Node* next;

    Node(string n, string p)
    {
        name = n;
        phone = p;
        next = NULL;
    }
};

class HashTable
{
    Node** table;
    int size;

public:
    HashTable(int s)
    {
        size = s;
        table = new Node*[size];

        for (int i = 0; i < size; i++)
        {
            table[i] = NULL;
        }
    }

    int hashFunction(string name)
    {
        int sum = 0;

        for (int i = 0; i < name.length(); i++)
        {
            sum += (unsigned char)name[i];
        }

        return sum % size;
    }

    void insert(string name, string phone)
    {
        int index = hashFunction(name);
        Node* newNode = new Node(name, phone);

        if (table[index] == NULL)
        {
            table[index] = newNode;
        }
        else
        {
            Node* temp = table[index];

            while (temp->next != NULL)
            {
                temp = temp->next;
            }

            temp->next = newNode;
        }

        cout << "Contact inserted at index "
             << index << endl;
    }

    void display()
    {
        cout << "\n===== TELEPHONE DIRECTORY =====\n";

        for (int i = 0; i < size; i++)
        {
            cout << i << ": ";

            Node* temp = table[i];

            if (temp == NULL)
            {
                cout << "NULL";
            }
            else
            {
                while (temp != NULL)
                {
                    cout << "[" << temp->name
                         << " : " << temp->phone << "]";

                    if (temp->next != NULL)
                    {
                        cout << " -> ";
                    }

                    temp = temp->next;
                }
            }

            cout << endl;
        }
    }

    void search(string name)
    {
        int index = hashFunction(name);
        Node* temp = table[index];

        while (temp != NULL)
        {
            if (temp->name == name)
            {
                cout << "\nContact Found!" << endl;
                cout << "Name: " << temp->name << endl;
                cout << "Phone: " << temp->phone << endl;
                return;
            }

            temp = temp->next;
        }

        cout << "\nContact Not Found!" << endl;
    }

    ~HashTable()
    {
        for (int i = 0; i < size; i++)
        {
            Node* temp = table[i];

            while (temp != NULL)
            {
                Node* nextNode = temp->next;
                delete temp;
                temp = nextNode;
            }
        }

        delete[] table;
    }
};

#include <iostream>
using namespace std;

struct Node {
    int data;
    Node* prev;
    Node* next;
};

void appendBit(Node*& head, Node*& tail, int bit) {
    Node* newNode = new Node();
    newNode->data = bit;
    newNode->next = NULL;
    newNode->prev = tail;

    if (head == NULL) {
        head = newNode;
    } else {
        tail->next = newNode;
    }
    tail = newNode;
}

void display(Node* head) {
    Node* temp = head;
    while(temp != NULL) {
        cout << temp->data;
        temp = temp->next;
    }
    cout << endl;
}

Node* getOnesComplement(Node* head) {
    Node* compHead = NULL;
    Node* compTail = NULL;
    Node* temp = head;

    while (temp != NULL) {
        int flippedBit = (temp->data == 0) ? 1 : 0;
        appendBit(compHead, compTail, flippedBit);
        temp = temp->next;
    }
    return compHead;
}

Node* getTowsComplement(Node* head) {
    Node* onesCompHead = getOnesComplement(head);
    Node* temp = onesCompHead;

    while (temp->next != NULL) {
        temp = temp->next;
    }

    int carry = 1;
    while (temp != NULL) {
        int sum = temp->data + carry;
        temp->data = sum % 2;
        carry = sum / 2;
        temp = temp->prev;
    }

    if (carry == 1) {
        Node* newHead = new Node();
        newHead->data = 1;
        newHead->prev = NULL;
        newHead->next = onesCompHead;
        onesCompHead->prev = newHead;
        onesCompHead = newHead;
    }

    return onesCompHead;
}

Node* addBinary(Node* head1, Node* head2) {
    Node* sumHead = NULL;
    Node* sumTail = NULL;
    Node* t1 = head1;
    Node* t2 = head2;

    while (t1 && t1->next) t1 = t1->next;
    while (t2 && t2->next) t2 = t2->next;

    int carry = 0;

    while (t1 != NULL || t2 != NULL || carry != 0) {
        int b1 = (t1 != NULL) ? t1->data : 0;
        int b2 = (t2 != NULL) ? t2->data : 0;
        int sum = b1 + b2 + carry;
        int bitResult = sum % 2;
        carry = sum / 2;

        Node* newNode = new Node();
        newNode->data = bitResult;
        newNode->prev = NULL;
        newNode->next = sumHead;

        if (sumHead != NULL) {
            sumHead->prev = newNode;
        }

        sumHead = newNode;

        if (t1) t1 = t1->prev;
        if (t2) t2 = t2->prev;
    }

    return sumHead;
}

#include <iostream>
#include <string>
using namespace std;

const int TABLE_SIZE = 10;

struct Client {
    string name;
    string phone;
    bool occupied;

    Client() {
        name = "";
        phone = "";
        occupied = false;
    }
};

class TelephoneDirectory {
private:
    Client table[TABLE_SIZE];

    // Hash function
    int hashFunction(string name) {
        int hash = 0;

        for (char ch : name) {
            hash = (hash * 31 + ch) % TABLE_SIZE;
        }

        return hash;
    }

public:

    // ---------------- LINEAR PROBING ----------------
    void insertLinear(string name, string phone) {
        int index = hashFunction(name);
        int start = index;

        while (table[index].occupied) {
            index = (index + 1) % TABLE_SIZE;

            // Table is full
            if (index == start) {
                cout << "Hash table is full!\n";
                return;
            }
        }

        table[index].name = name;
        table[index].phone = phone;
        table[index].occupied = true;

        cout << "Client inserted at index " << index << endl;
    }

    // ---------------- QUADRATIC PROBING ----------------
    void insertQuadratic(string name, string phone) {
        int index = hashFunction(name);

        for (int i = 0; i < TABLE_SIZE; i++) {
            int newIndex = (index + i * i) % TABLE_SIZE;

            if (!table[newIndex].occupied) {
                table[newIndex].name = name;
                table[newIndex].phone = phone;
                table[newIndex].occupied = true;

                cout << "Client inserted at index "
                     << newIndex << endl;
                return;
            }
        }

        cout << "Hash table is full!\n";
    }

    // Search using linear probing
    void searchLinear(string name) {
        int index = hashFunction(name);
        int start = index;

        while (table[index].occupied) {
            if (table[index].name == name) {
                cout << "Telephone number of " << name
                     << " : " << table[index].phone << endl;
                return;
            }

            index = (index + 1) % TABLE_SIZE;

            if (index == start)
                break;
        }

        cout << "Client not found!\n";
    }

    // Search using quadratic probing
    void searchQuadratic(string name) {
        int index = hashFunction(name);

        for (int i = 0; i < TABLE_SIZE; i++) {
            int newIndex = (index + i * i) % TABLE_SIZE;

            if (!table[newIndex].occupied)
                break;

            if (table[newIndex].name == name) {
                cout << "Telephone number of " << name
                     << " : " << table[newIndex].phone << endl;
                return;
            }
        }

        cout << "Client not found!\n";
    }

    // Display hash table
    void display() {
        cout << "\n--- Telephone Directory ---\n";

        for (int i = 0; i < TABLE_SIZE; i++) {
            cout << "Index " << i << " : ";

            if (table[i].occupied)
                cout << table[i].name << " -> "
                     << table[i].phone;
            else
                cout << "Empty";

            cout << endl;
        }
    }
};
